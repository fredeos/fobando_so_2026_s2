#include <string.h>

#include "esp_log.h"

#include "common/protocol.h"
#include "serial/esp32.h"
#include "serial/serial.h"
#include "gpio/gpio.h"

// << Board pins >>
#define LED0 26
#define LED1 25
#define LED2 33
#define LED3 32

// << Program constants >>
#define UART_PORT   UART_NUM_0
#define BAUD_RATE   115200

#define MESSAGE_LEN 128u

#define PING_EVENT_CODE  2u
#define CHECK_EVENT_CODE 4u

void app_main(void){
    // << Configure UART port >>
    esp32_uart_t uart = {
        .port_num = UART_PORT,
        .ticks = pdMS_TO_TICKS(100)
    };

    // UART0 is commonly connected to the DevKit USB-UART bridge.
    // GPIO pins are left at their existing UART assignments.
    const uart_config_t config = {
        .baud_rate = BAUD_RATE,
        .data_bits = UART_DATA_8_BITS,
        .parity = UART_PARITY_DISABLE,
        .stop_bits = UART_STOP_BITS_1,
        .flow_ctrl = UART_HW_FLOWCTRL_DISABLE,
        .source_clk = UART_SCLK_DEFAULT
    };

    const int uart_buffer_size = (1024 * 1);
    ESP_ERROR_CHECK(
        uart_driver_install(uart.port_num, uart_buffer_size, uart_buffer_size, 10, NULL, 0)
    );

    ESP_ERROR_CHECK(
        uart_param_config(uart.port_num, &config)
    );

    // << Disable logs >>
    // Keep application logs off the serial protocol channel.
    esp_log_level_set("*", ESP_LOG_NONE);

    // << Configure GPIO >>
    board_gpio_init(LED0, GPIO_OUTPUT);
    board_gpio_init(LED1, GPIO_OUTPUT);
    board_gpio_init(LED2, GPIO_OUTPUT);
    board_gpio_init(LED3, GPIO_OUTPUT);

    // << Configure procedure variables >
    // setup buffers
    uint8_t in [PROTOCOL_MAX_FRAME];
    serial_packet_t input = {
        .mode = SERIAL_IN,
        .buffer = in,
        .length = sizeof(in),
        .processed = 0,
        .complete  = false
    };

    uint8_t out[PROTOCOL_MAX_FRAME];
    serial_packet_t output = {
        .mode = SERIAL_OUT,
        .buffer = out,
        .length = sizeof(out),
        .processed = 0,
        .complete  = false
    };
    int  event_code   = 0;
    bool msg_pending  = false;
    // set basic messages
    const char pong[]        = "PONG from ESP32";
    const char heartbeat[]   = "HEARTBEAT from ESP32";
    const char default_msg[] = "HEARTBEAT";

    // << Procedure loop >>
    while (1) {
        // ----------------------[Loop start]----------------------
        board_gpio_write(LED0, GPIO_LOW);
        board_gpio_write(LED1, GPIO_LOW);
        board_gpio_write(LED2, GPIO_LOW);
        board_gpio_write(LED3, GPIO_LOW);
        // ----------------------[Read serial]----------------------
        // [Read bytes]
        size_t ser_rd_bytes = 0;
        serial_errno_t ser_rd_errno = serial_read(
            &uart,
            &input,
            &ser_rd_bytes
        );
        if (ser_rd_errno != SERIAL_OK){ // restart on error
            input.processed = 0;
            input.complete  = false;
        }
        // [Check IF complete]
        protocol_check_t frame = {
            .is_valid = false,
            .is_valuable = false
        };
        if (input.complete){
            // + Check IF line has correct protocol structure
            protocol_errno_t frame_errno = protocol_validate_bytes(
                input.buffer,
                input.length,
                &frame
            );
        }
        if (frame.is_valid)
            board_gpio_write(LED0, GPIO_HIGH);
        // [Check IF valuable]
        if (frame.is_valuable && event_code == 0){
            char message[MESSAGE_LEN];
            // + Extract data to a string
            size_t rd_bytes = 0;
            protocol_errno_t rd_errno = protocol_extract_bytes(
                message,
                sizeof(message),
                input.buffer,
                input.length,
                PROTOCOL_HEADER_BYTES,
                &rd_bytes
            );
            // + Identify event
            if (rd_errno == PROTOCOL_OK && rd_bytes > 0){
                if (strcmp(message, "PING") == 0){
                    event_code = PING_EVENT_CODE;
                }
                else if (strcmp(message, "CHECK") == 0) {
                    event_code = CHECK_EVENT_CODE;
                }
            }
        }
        // ----------------------[Write serial]----------------------
        // [Configure output message]
        if (!msg_pending){
            // + Configure payload
            const char* response = default_msg;
            if (event_code == PING_EVENT_CODE){
                response = pong;
                board_gpio_write(LED1, GPIO_HIGH);
            }
            else if (event_code == CHECK_EVENT_CODE){
                response = heartbeat;
                board_gpio_write(LED2, GPIO_HIGH);
            }
            // + Configure header & footer
            uint8_t header = PROTOCOL_MAGIC_1;
            uint8_t footer = PROTOCOL_MAGIC_3;
            if (event_code > 0){
                header = PROTOCOL_MAGIC_0;
                footer = PROTOCOL_MAGIC_2;
            }
            // + Configure buffer
            size_t wd_bytes = 0;
            protocol_errno_t wd_errno = protocol_load_bytes(
                output.buffer,
                output.length,
                response,
                strlen(response),
                header,
                footer,
                0x00u,
                &wd_bytes
            );
            if (wd_errno == PROTOCOL_OK && wd_bytes > 0)
                msg_pending = true;
        }
        // [Send current messgae]
        if (msg_pending){
            // + Send bytes
            size_t ser_wd_bytes = 0;
            serial_errno_t ser_wd_errno = serial_write(
                &uart,
                &output,
                &ser_wd_bytes
            );
            if (ser_wd_errno != SERIAL_OK){ // restart on error
                output.processed = 0;
                output.complete  = false;
            }
            // + Check IF completed
            if (output.complete){
                board_gpio_write(LED3, GPIO_HIGH);
                msg_pending = false;
                event_code = 0;
            }
        }
    }
}
